\section*{Recommended Textbooks}

\begin{enumerate}
  \item Halliday, D., Resnick, R., and Walker, J.
  \textit{Fundamentals of Physics} (10th ed.).
  Wiley. --- The gold standard university physics textbook.
  Use for deeper mathematical treatment.

  \item Serway, R.A., and Jewett, J.W.
  \textit{Physics for Scientists and Engineers} (9th ed.).
  Cengage Learning. --- Excellent derivations and
  comprehensive coverage.

  \item Giancoli, D.C.
  \textit{Physics: Principles with Applications} (7th ed.).
  Pearson. --- Particularly strong on real-world applications.

  \item Nelkon, M., and Parker, P.
  \textit{Advanced Level Physics} (7th ed.).
  Heinemann. --- Widely used in Nigerian secondary schools;
  good for bridging WAEC/UTME to university level.
\end{enumerate}

\section*{Online Resources}

\begin{enumerate}
  \item Khan Academy Physics (\texttt{khanacademy.org}) ---
  Free video explanations of all topics in this textbook.

  \item HyperPhysics (\texttt{hyperphysics.phy-astr.gsu.edu})
  --- Comprehensive concept maps and quick references.

  \item PhET Interactive Simulations
  (\texttt{phet.colorado.edu}) --- Free physics simulations
  for visualising concepts that are difficult to observe
  directly.
\end{enumerate}
